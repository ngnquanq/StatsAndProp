\section{Experimental Results}

\begin{frame}{Experiment Setup}
    \textbf{Datasets:}
    \begin{itemize}
        \item \textbf{Synthetic 2D Blobs:} 3 clusters, varying densities
        \item \textbf{Iris Dataset:} 150 samples, 4 features, 3 species
        \item \textbf{Wine Dataset:} 178 samples, 13 features, 3 types
        \item \textbf{Image Segmentation:} Color-based clustering
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Evaluation Metrics:}
    \begin{itemize}
        \item Silhouette Score (cluster quality)
        \item Adjusted Rand Index (vs. ground truth)
        \item Computation time
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Comparison:}
    \begin{itemize}
        \item K-means
        \item DBSCAN
    \end{itemize}
\end{frame}

\begin{frame}{Results: Wine Dataset}
    \textbf{Challenge:} High-dimensional (13 features)

    \vspace{0.3cm}

    \begin{table}
        \centering
        \begin{tabular}{lcccc}
            \toprule
            \textbf{Method} & \textbf{Clusters Found} & \textbf{ARI} & \textbf{Silhouette} & \textbf{Time (s)} \\
            \midrule
            K-means (k=3) & 3 & 0.37 & 0.28 & 0.03 \\
            DBSCAN & 4 & 0.21 & 0.15 & 0.12 \\
            Mean Shift (raw) & 7 & 0.19 & 0.22 & 1.24 \\
            \textcolor{customred}{Mean Shift (PCA)} & \textcolor{customred}{3} & \textcolor{customred}{0.41} & \textcolor{customred}{0.35} & 0.45 \\
            \bottomrule
        \end{tabular}
    \end{table}

    \vspace{0.3cm}

    \textbf{Key Insight:}
    \begin{itemize}
        \item Raw high-dim data → over-segmentation (7 clusters)
        \item PCA projection (3 components) → correct clustering
        \item \textcolor{customred}{Demonstrates curse of dimensionality effect}
    \end{itemize}
\end{frame}

\begin{frame}{Bandwidth Sensitivity Analysis}
    \begin{center}
        \textit{[Insert plot: Number of clusters vs. bandwidth]}

        \vspace{0.3cm}

        \textit{X-axis: Bandwidth $h$ | Y-axis: Number of clusters}
    \end{center}

    \vspace{0.5cm}

    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Small $h$ ($h < 0.5$):}
            \begin{itemize}
                \item Many tiny clusters
                \item Over-segmentation
                \item High sensitivity to noise
            \end{itemize}
        \end{column}

        \begin{column}{0.5\textwidth}
            \textbf{Large $h$ ($h > 2.0$):}
            \begin{itemize}
                \item Few large clusters
                \item Under-segmentation
                \item Misses subtle structure
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.3cm}

    \textcolor{customred}{\textbf{Recommendation:}} Use cross-validation or Scott's rule: $h \approx n^{-1/(d+4)}$
\end{frame}
